# Tạo Validation

Validation được dùng để validate dữ liệu trước khi gửi lên sever

Ví dụ: Tạo 1 file check validate cho form login


```tsx
import * as yup from 'yup';
import { basePasswordValidator, baseStringValidator } from '../base';
import { t } from 'i18next';

export const loginValidationSchema = yup.object({
  username: baseStringValidator({
    required: true,
    label: t('sign_in.username'),
    minLength: 6,
  }),
  password: basePasswordValidator({
    required: true,
    label: t('sign_in.password'),
    minLength: 6,
  }),
});

```

# B. Cách sử dụng

```tsx
'use client';
import BaseButton from '@/components/common/BaseButton';
import FormContainer from '@/components/form/FormContainer';
import FormTextInput from '@/components/form/FormTextInput';
import { loginValidationSchema } from '@/validators/auth/login.schema';
import { Typography } from 'antd';
import { useTranslation } from 'react-i18next';

export default function Login() {
  const { t } = useTranslation();

  return (
    <div className="flex items-center justify-center min-h-screen bg-gray-100 px-4">
      <div className="bg-white p-8 rounded-2xl shadow-md w-full max-w-md">
        <Typography.Title level={2} className="text-center mb-6">
          {t('common.sign-in')}
        </Typography.Title>
        <FormContainer
          defaultValues={{ username: '', password: '' }}
          onSubmit={values => {
            values;
          }}
          defaultFocus="password"
          resolver={loginValidationSchema}
        >
          {({ handleSubmit }) => (
            <div>
              <FormTextInput
                name="username"
                label={t('sign_in.username')}
                placeholder="Nhập tên người dùng"
                required
              />
              <FormTextInput
                name="password"
                label={t('sign_in.password')}
                placeholder="Nhập mật khẩu"
              />
              <BaseButton onClick={handleSubmit}>
                {t('sign_in.sign_in')}
              </BaseButton>
            </div>
          )}
        </FormContainer>
      </div>
    </div>
  );
}

```
